# MT_ENABLE_IMGUI_TEST_ENGINE is NOT here. It is a capability flag -- mtcaps
# derives it from MT_CAP_TEST_ENGINE -- and mt_apply_capabilities below supplies
# it. Hardcoding =1 here made this file a SECOND source of truth that outranked
# the manifest: a directory definition lands on the command line before the
# target one, so `-DMT_ENABLE_IMGUI_TEST_ENGINE=1 ... =0` compiled with the
# manifest's value only by luck of last-wins, and the -Wno-macro-redefined below
# is what kept the collision quiet. See the fallback after the fragment include.

    -DIMGUI_IMPL_OPENGL_LOADER_GL3W

# ---------------------------------------------------------------------------
# Capabilities
#
# MT_CAPS_FRAGMENT is generated by MTEngineSDL/tools/mtcaps into $MT_OUT --
# outside every checkout -- and passed in by this app's build-linux.sh.
#
# The app MUST include it, not just the engine. Under raw-archive linkage the
# engine is configured by a SEPARATE cmake process that runs first, so a fragment
# included only there sets the engine's defines and reaches this app's compile
# line not at all -- and the app would then compile against a different
# capability set than the archive it links. That is the httplib ODR shape, one
# layer up.
#
# MT_ENGINE_BUILD_DIR is where that separate process put libMTEngineSDL.a. It
# used to be a fixed MTEngineSDL/build shared by all four apps, which is both a
# write inside the engine checkout and the collision that makes two concurrent
# app builds unsafe however well the archives are keyed.
# ---------------------------------------------------------------------------
if (DEFINED MT_CAPS_FRAGMENT)
    if (NOT EXISTS "${MT_CAPS_FRAGMENT}")
        message(FATAL_ERROR
            "MT_CAPS_FRAGMENT=${MT_CAPS_FRAGMENT} does not exist.\n"
            "Build through ./build-linux.sh, which generates it.")
    endif()
    include("${MT_CAPS_FRAGMENT}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MT_CAPS_FRAGMENT}")
    message(STATUS "Capabilities: ${MT_CAPS_RESOLVED}")
else()
    # Fallback for a bare `cmake ../` with no wrapper run, matching what
    # MTEngineSDLDummyAppCaps.xcconfig does with MT_APP_DEFAULT_CAPS_NO on macOS.
    # The configure below still fails on MT_CAPS_LIBS_DIR -- that is the designed
    # outcome -- but the app's guards are value-style `#if`, so leaving the flag
    # undefined here would silently mean "test engine off" rather than "no
    # capability set was supplied".
    add_definitions(-DMT_ENABLE_IMGUI_TEST_ENGINE=1)
endif()

if (NOT DEFINED MT_ENGINE_BUILD_DIR)
    # The engine's CMake tree left the engine checkout, so MTEngineSDL/build no
    # longer receives anything -- defaulting there would send a manual `cmake ../`
    # looking for an archive in a directory nothing writes, and surface as an
    # opaque link error rather than a useful one.
    #
    # The default is where a STANDALONE engine build actually puts it. An app
    # built through ./build-linux.sh gets the keyed $MT_OUT path instead, passed
    # in as -DMT_ENGINE_BUILD_DIR.
    if (DEFINED ENV{MTENGINE_BUILD_ROOT})
        set(MT_BUILD_ROOT "$ENV{MTENGINE_BUILD_ROOT}")
    elseif (DEFINED ENV{XDG_CACHE_HOME})
        set(MT_BUILD_ROOT "$ENV{XDG_CACHE_HOME}/mtengine")
    else()
        set(MT_BUILD_ROOT "$ENV{HOME}/.cache/mtengine")
    endif()
    set(MT_ENGINE_BUILD_DIR
        "${MT_BUILD_ROOT}/_standalone/linux/${CMAKE_SYSTEM_PROCESSOR}/Release/engine-build")
endif()

# A missing engine archive is a hard error naming the command that fixes it, not
# a link failure fifty lines later. There is no downgrade tier.
if (NOT EXISTS "${MT_ENGINE_BUILD_DIR}/libMTEngineSDL.a")
    message(FATAL_ERROR
        "libMTEngineSDL.a not found in ${MT_ENGINE_BUILD_DIR}.\n"
        "Build through ./build-linux.sh, which builds the engine against this "
        "app's mtengine.caps and passes -DMT_ENGINE_BUILD_DIR.")
endif()

# ---------------------------------------------------------------------------
# Engine archives
#
# libMTEngineSDL.a is a STATIC library, so every archive MTEngineSDL links
# PRIVATE (llama.cpp, the codec bundles, FTXUI, mbedTLS) stays unresolved until
# the final executable link. Apps that link libMTEngineSDL.a by raw archive path
# — rather than as an imported CMake target — get no transitive propagation and
# must repeat the list here.
# ---------------------------------------------------------------------------
set(MT_ENGINE_LIB      "${MT_ENGINE_BUILD_DIR}/libMTEngineSDL.a")
set(MT_LLAMA_CPP_LIB   "${MT_ENGINE_BUILD_DIR}/libllama_cpp.a")
# Where the dependency archives are. Outside every checkout, keyed by the
# resolved capability set -- and ASKED OF THE ENGINE rather than resolved here,
# because <backend> is a component of that path and the engine wrapper is what
# decides MT_GGML_NATIVE from `uname -m`. An app resolving for itself would get
# backend=default while the engine populated backend=<hash>, link an empty
# directory, and fail with undefined llama symbols that look like a broken
# project. See MTEngineSDL/build-linux.sh --print-deps-dir.
if(NOT MT_CAPS_LIBS_DIR)
    message(FATAL_ERROR
        "MT_CAPS_LIBS_DIR is not set. Build through ./build-linux.sh, which asks "
        "the engine wrapper for the keyed dependency directory and passes it in.")
endif()

set(MT_VIDEO_CODECS_LIB "${MT_CAPS_LIBS_DIR}/libmt_video_codecs.a")
set(MT_FTXUI_LIB       "${MT_CAPS_LIBS_DIR}/libftxui.a")

if(NOT EXISTS "${MT_ENGINE_LIB}")
    message(FATAL_ERROR "MTEngineSDL not built: ${MT_ENGINE_LIB}\n"

endif()

# Capability-gated archives.
#
# These used to be linked unconditionally, which means "no llama.cpp code
# compiled" still shipped 598 MB of archive -- and this repo is the host whose
# acceptance criterion is "on and off for each capability", so it could not
# demonstrate `off` at all while linking the archive regardless.
set(MT_CAP_GATED_LIBS "")
if(NOT DEFINED MT_ENABLE_LLAMA_CPP OR MT_ENABLE_LLAMA_CPP)
    list(APPEND MT_CAP_GATED_LIBS "${MT_LLAMA_CPP_LIB}")
endif()
if(NOT DEFINED MT_ENABLE_MBEDTLS OR MT_ENABLE_MBEDTLS)
    list(APPEND MT_CAP_GATED_LIBS "${MT_MBEDTLS_LIB}")
endif()
if(NOT DEFINED MT_ENABLE_FTXUI OR MT_ENABLE_FTXUI)
    list(APPEND MT_CAP_GATED_LIBS "${MT_FTXUI_LIB}")
endif()
if(NOT DEFINED MT_ENABLE_UWEBSOCKETS OR MT_ENABLE_UWEBSOCKETS)
    list(APPEND MT_CAP_GATED_LIBS "${MT_USOCKETS_LIB}")
endif()

# libheif is a SYSTEM package on Linux (see MTEngineSDL/CMakeLists.txt's own
# pkg_check_modules(HEIF ...)), not one of the archives MTEngineSDL/build-linux.sh
# produces -- so it is neither bundled into MT_IMAGE_CODECS_LIB nor covered by the
# loop above. The engine links it PRIVATE into libMTEngineSDL.a, which resolves
# nothing for a raw-archive consumer (see the comment at the top of this file):
# this app must ask pkg-config for it independently, exactly as the engine did,
# or the final link fails with undefined heif_* references. Safe to mark
# REQUIRED: if MT_ENABLE_LIBHEIF is on, the engine's own build already hard-erred
# before this app's cmake ever ran had libheif been missing.
if(NOT DEFINED MT_ENABLE_LIBHEIF OR MT_ENABLE_LIBHEIF)
    pkg_check_modules(HEIF REQUIRED libheif)
    include_directories(${HEIF_INCLUDE_DIRS})
    list(APPEND MT_CAP_GATED_LIBS ${HEIF_LIBRARIES})
endif()
